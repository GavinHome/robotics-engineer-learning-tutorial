```cpp
const int LED_PIN = 2;
const int POT_PIN = 34;

void setup() {
  pinMode(LED_PIN, OUTPUT);
  ledcSetup(0, 5000, 8);  // 通道0, 5kHz, 8位分辨率
  ledcAttachPin(LED_PIN, 0);
  Serial.begin(115200);
}

void loop() {
  int potValue = analogRead(POT_PIN);  // 0-4095 (12位ADC)
  int brightness = map(potValue, 0, 4095, 0, 255);
  ledcWrite(0, brightness);
  Serial.printf("Pot: %d, Brightness: %d\n", potValue, brightness);

}
```

```cpp
const int VOLTAGE_PIN = 34;
const float VREF = 3.3;
const int ADC_MAX = 4095;
const float R1 = 1000.0;  // 下拉电阻 kΩ
const float R2 = 10000.0; // 上拉电阻 kΩ

void setup() {
  Serial.begin(115200);
  analogReadResolution(12);
}

void loop() {
  long sum = 0;
  for (int i = 0; i < 10; i++) {
    sum += analogRead(VOLTAGE_PIN);
    delay(5);
  }
  float avg = sum / 10.0;
  float vout = avg * VREF / ADC_MAX;
  float vin = vout * (R1 + R2) / R1;

  Serial.printf("Input: %.2fV (ADC: %.0f)\n", vin, avg);
  delay(1000);
}
```

### Day 17：HC-SR04 超声波传感器
**目标**：读取距离数据，理解时序通信

**任务**：
1. HC-SR04 原理：
   - Trig：触发引脚（发送 10μs 高电平）
   - Echo：回响引脚（返回高电平，宽度 = 距离 × 58μs）
   - 测量范围：2cm-400cm
2. 接线：
   - VCC → 5V（注意：ESP32 是 3.3V，但 HC-SR04 需要 5V）
   - GND → GND
   - Trig → GPIO5
   - Echo → GPIO18（需电平转换！5V 输出可能损坏 ESP32）
3. 代码实现距离测量

**代码参考**：
```cpp
const int TRIG_PIN = 5;
const int ECHO_PIN = 18;

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
}

float readDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH, 30000);  // 超时 30ms
  float distance = duration * 0.0343 / 2;  // cm
  return distance;
}

void loop() {
  float d = readDistance();
  if (d > 0 && d < 400) {
    Serial.printf("Distance: %.1f cm\n", d);
  } else {
    Serial.println("Out of range");
  }
  delay(500);
}
```

**输出**：距离测量实验数据 + HC-SR04 时序图

### Day 18：MPU-6050 IMU 传感器
**目标**：读取加速度计和陀螺仪数据，理解 I2C 通信

**任务**：
1. MPU-6050 简介：
   - 三轴加速度计 + 三轴陀螺仪
   - I2C 地址：0x68（AD0=GND）或 0x69（AD0=VCC）
2. 接线（I2C）：
   - VCC → 3.3V（或 5V，模块通常有稳压）
   - GND → GND
   - SCL → GPIO22（ESP32-S3 默认 I2C SCK）
   - SDA → GPIO21（ESP32-S3 默认 I2C SDA）
3. 使用 Adafruit MPU6050 库
4. 读取加速度 X/Y/Z 和陀螺仪 X/Y/Z
5. 计算倾斜角度（简化版，仅用加速度计）

**代码参考**：
```cpp
#include <Wire.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>

Adafruit_MPU6050 mpu;

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);  // SDA, SCL

  if (!mpu.begin()) {
    Serial.println("MPU6050 not found!");
    while (1);
  }

  mpu.setAccelerometerRange(MPU6050_RANGE_8_G);
  mpu.setGyroRange(MPU6050_RANGE_500_DEG);
  mpu.setFilterBandwidth(MPU6050_BAND_21_HZ);

  Serial.println("MPU6050 Ready");
}

void loop() {
  sensors_event_t a, g, t;
  mpu.getEvent(&a, &g, &t);

  Serial.printf("Accel: X=%.2f Y=%.2f Z=%.2f m/s²\n",
```

   - 反转 2 秒
   - 刹车 1 秒

```cpp
const int AIN1 = 25;
const int AIN2 = 26;
const int PWMA = 27;
const int STBY = 14;

void setup() {
  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);
  pinMode(PWMA, OUTPUT);
  pinMode(STBY, OUTPUT);
  digitalWrite(STBY, HIGH);  // 使能驱动
  Serial.begin(115200);
}

void setMotor(bool dir, int speed) {
  // dir: true=正转, false=反转
  digitalWrite(AIN1, dir ? HIGH : LOW);
  digitalWrite(AIN2, dir ? LOW : HIGH);
  ledcWrite(0, speed);  // PWM 通道0
}

void setupPWM() {
  ledcSetup(0, 5000, 8);
  ledcAttachPin(PWMA, 0);
}

void loop() {
  // 正转加速
  for (int s = 0; s <= 255; s += 5) {
    setMotor(true, s);
    delay(50);
  }
  delay(1000);

  // 反转
  for (int s = 255; s >= 0; s -= 5) {
    setMotor(false, s);
    delay(50);
  }
  delay(1000);

  // 刹车
  digitalWrite(AIN1, LOW);
  digitalWrite(AIN2, LOW);
  delay(2000);
}
```

```cpp
const int SERVO_PIN = 18;

void setup() {
  Serial.begin(115200);
  ledcSetup(0, 50, 16);     // 通道0, 50Hz, 16位
  ledcAttachPin(SERVO_PIN, 0);
  Serial.println("Enter angle (0-180):");
}

void loop() {
  if (Serial.available()) {
    int angle = Serial.parseInt();
    if (angle >= 0 && angle <= 180) {
      int duty = (angle / 180.0) * 2000 + 500;  // 0.5ms-2.5ms
      ledcWrite(0, duty * 65536 / 20000);  // 转换为16位占空比

  }
```

```cpp
class Motor {
public:
  Motor(int in1, int in2, int pwm, int stby)
    : _in1(in1), _in2(in2), _pwm(pwm), _stby(stby) {}

  void begin() {
    pinMode(_in1, OUTPUT);
    pinMode(_in2, OUTPUT);
    ledcSetup(_channel, 5000, 8);
    ledcAttachPin(_pwm, _channel);
    pinMode(_stby, OUTPUT);
    digitalWrite(_stby, HIGH);
  }

  void drive(int speed) {  // -255 to 255
    if (speed > 0) {
      digitalWrite(_in1, HIGH);
      digitalWrite(_in2, LOW);
    } else {
      digitalWrite(_in1, LOW);
      digitalWrite(_in2, HIGH);
    }
    ledcWrite(_channel, abs(speed));
  }

  void stop() {
    digitalWrite(_in1, LOW);
    digitalWrite(_in2, LOW);
  }

private:
  int _in1, _in2, _pwm, _stby;
  int _channel = 0;
};

// 使用

Motor rightMotor(32, 33, 4, 14);

void setup() {

  rightMotor.begin();
  Serial.begin(115200);
}

void loop() {
  float dist = readDistance();
  if (dist > 0 && dist < 15) {
    // 避障：后退 → 右转 → 前进
    leftMotor.drive(-150);
    rightMotor.drive(-150);
    delay(500);
    leftMotor.drive(200);
    rightMotor.drive(-200);
    delay(300);
  } else {
    leftMotor.drive(150);
    rightMotor.drive(150);
  }

}
```

## 附录：常见问题与排障

### ESP32-S3 常见问题

**Q: 上传失败，提示 "A fatal error occurred: Failed to connect to ESP32-S3"**
- 检查 USB 线是否支持数据传输（某些充电线只有电源）
- 按住 BOOT 键，点击上传，释放 BOOT
- 检查串口驱动（CP210x 或 CH340）

**Q: 串口输出乱码**
- 检查波特率是否一致（代码和串口监视器都要 115200）
- 检查接地是否良好

**Q: ADC 读数不稳定**
- 多次采样取平均
- 在 VREF 引脚加滤波电容
- 远离 Wi-Fi 射频干扰

**Q: Wi-Fi 连接失败**
- ESP32 仅支持 2.4GHz Wi-Fi
- 检查 SSID 是否含中文（可能有问题）
- 增加连接超时时间

### 焊接常见问题

**Q: 焊点呈球状、不湿润**
- 烙铁温度不够
- 焊面氧化（需要打磨或加助焊剂）

**Q: 虚焊**
- 加热时间不足
- 焊锡量不够

**Q: 短路**
- 焊锡过多
- 未注意相邻引脚间距
